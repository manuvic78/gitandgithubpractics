git init
#para iniciar el repositorio

git remote add origin <repository_url>
#agregar el origen del repositorio

git branch -M main
#ver la rama y cambiarle el nombre 

git add .
#agregar un archivo con su debida extension 

git commit -m 'Your commit message here!'
#enviar un comentario con el debido cambio 

git status
#te habla del estatus actual de tu repositorio 

git push -u origin main
#envia los cambios con sus debidos commits a la rama principal
